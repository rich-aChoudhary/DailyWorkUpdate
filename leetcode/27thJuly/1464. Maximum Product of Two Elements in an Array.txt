class Solution:
    def maxProduct(self, nums: List[int]) -> int:
        first,second=nums[0],nums[1]
        for i in range(2,len(nums)):
            if nums[i]>first:
                second=max(first,second)
                first=nums[i]
            elif nums[i]>second:
                second=nums[i]

        return (first-1)*(second-1)
