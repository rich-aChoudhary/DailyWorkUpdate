class Solution:
    def predictTheWinner(self, nums: List[int]) -> bool:
        def solve(i,j):
            if i==j:
                return nums[i]
            take_i=nums[i]-solve(i+1,j)
            take_j=nums[j]-solve(i,j-1)
            return max(take_i,take_j)
        return True if solve(0,len(nums)-1)>=0 else False
